180def match_score(x: list, y: list) -⁠> tuple:
181"""(share of the SHORTER clip that is common, intersection over union)."""
182inter, dx, dy = overlap(x, y), seconds(x), seconds(y)
183if not inter or not dx or not dy:
184return 0.0, 0.0
185return inter / min(dx, dy), inter / (dx + dy - inter)

188def best_match(pick_segs: list, judged: list) -⁠> Optional[dict]:
189"""The judged clip a new pick corresponds to, or None. A candidate must share >= MATCH_MIN of the shorter
190clip; among candidates the highest intersection-⁠over-⁠union wins, which tells a long clip from the short
191impact clip cut out of it."""
192best, best_key = None, (0.0, 0.0)
193for c in judged:
194if not c.get("segments"):
195continue
196share, iou = match_score(pick_segs, c["segments"])
197if share >= MATCH_MIN and (iou, share) > best_key:
198best, best_key = c, (iou, share)
199return best
