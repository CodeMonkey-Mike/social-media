① Lane 1 — longform → POST graph (segment 7, one platform per invocation)

flowchart TD
  C1((("1"))) --> Q1[/"longs.json entry pending"/]
  Q1 --> P1{"Mike authorizes posting?
(the invocation IS the decision)"}
  P1 -- "run.py post --platform p
SEQUENTIAL, x3 platforms" --> PFD{"front door: platform in
rumble/bitchute/facebook?
pending entry? mp4 on disk?"}
  P1 -- "not yet" --> WAIT(["stays pending"]):::halt
  PFD -- "no" --> PREF(["refused before Chrome opens"]):::halt
  PFD -- "yes" --> p_post

  subgraph GP["POST StateGraph — run.py post · segment 7"]
    direction TB
    p_post["post — shells upload_longform_<p>.py
(uploader does its own title-keyed
longs.json write-back)"]:::wnode --> p_vpost["verify_post — re-reads longs.json
FROM DISK: status flipped + url?"]:::vnode
  end
  p_post -. "exit != 0: halt (READ THE LOG —
the post may have landed; never relaunch on reflex)" .-> hp(["failed run"]):::halt
  p_vpost -. "no flip: halt" .-> hp

  p_vpost --> PLIVE{"liveness verified
by the uploader?"}
  PLIVE -- "yes" --> POK(["posted + url"]):::done
  PLIVE -- "no" --> PUV(["posted_unverified — Mike checks
the channel; NEVER re-run"]):::done

  POK --> NEXTP{"more platforms
pending?"}
  NEXTP -- "yes: next invocation" --> P1
  NEXTP -- "no" --> DONE1(["LANE 1 done: Rumble ·
BitChute · Facebook live"]):::done
